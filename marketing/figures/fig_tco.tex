    table[x=year, y=legacy] {figures/data/tco.dat} \closedcycle;
\addplot[seriesRatio, line width=1.3pt, mark=square*, mark size=2pt,

    table[x=year, y=ratio] {figures/data/tco.dat} \closedcycle;
